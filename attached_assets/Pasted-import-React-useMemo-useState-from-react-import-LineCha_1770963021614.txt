type MasteryStatus = "red" | "amber" | "green";

type TrendPoint = {
  label: string; // e.g., "Exam 1"
  score: number; // percent
};

type Topic = {
  id: string;
  name: string;
  strand: "Environments" | "Ventures" | "Roles" | "Operations";
  mastery: number; // 0..100
  probability: number; // 0..1 (trend weight)
};

type SubjectSnapshot = {
  subjectId: string;
  subjectName: string;
  grade: number;
  overallMastery: number; // 0..100
  lastExamScore: number; // 0..100
  avgMcq: number;
  avgCases: number;
  avgEssays: number;
  integrityScore: number; // 0..100
  strands: Record<"Environments" | "Ventures" | "Roles" | "Operations", number>; // 0..100
  topics: Topic[];
  trend: TrendPoint[];
  timeThisWeekMins: number;
};

type Badge = {
  id: string;
  title: string;
  description: string;
  earned: boolean;
  earnedAt?: string;
  icon: string; // emoji for now
};

type WeeklyPlanItem = {
  day: "Mon" | "Tue" | "Wed" | "Thu" | "Fri" | "Sat" | "Sun";
  label: string;
  durationMins: number;
  marks?: number;
  completed: boolean;
};

function statusFromMastery(m: number): MasteryStatus {
  if (m < 60) return "red";
  if (m < 85) return "amber";
  return "green";
}

function statusPill(status: MasteryStatus) {
  switch (status) {
    case "green":
      return "bg-emerald-500/15 text-emerald-300 border-emerald-500/30";
    case "amber":
      return "bg-amber-500/15 text-amber-300 border-amber-500/30";
    case "red":
      return "bg-rose-500/15 text-rose-300 border-rose-500/30";
  }
}

function clampPct(n: number) {
  return Math.max(0, Math.min(100, Math.round(n)));
}

function fmtPct(n: number) {
  return `${clampPct(n)}%`;
}

function readinessScore(overall: number, highProbAvg: number, recentTrendAvg: number) {
  // Exam readiness composite
  return clampPct(overall * 0.5 + highProbAvg * 0.3 + recentTrendAvg * 0.2);
}

function calcHighProbabilityAvg(topics: Topic[], threshold = 0.85) {
  const hp = topics.filter((t) => t.probability >= threshold);
  if (!hp.length) return 0;
  return hp.reduce((a, t) => a + t.mastery, 0) / hp.length;
}

function calcRecentTrendAvg(trend: TrendPoint[], lastN = 3) {
  const slice = trend.slice(-lastN);
  if (!slice.length) return 0;
  return slice.reduce((a, p) => a + p.score, 0) / slice.length;
}

function riskLabel(integrityScore: number) {
  if (integrityScore >= 95) return { label: "Clean", cls: "text-emerald-300" };
  if (integrityScore >= 80) return { label: "Monitor", cls: "text-amber-300" };
  return { label: "High Risk", cls: "text-rose-300" };
}

function Tile({ label, value, sub }: { label: string; value: string; sub?: string }) {
  return (
    <div className="rounded-2xl border border-slate-700/60 bg-slate-900/40 p-4 shadow-sm">
      <div className="text-xs text-slate-300/80">{label}</div>
      <div className="mt-1 text-2xl font-semibold text-slate-100">{value}</div>
      {sub ? <div className="mt-1 text-xs text-slate-400">{sub}</div> : null}
    </div>
  );
}

function ProgressBar({ label, value }: { label: string; value: number }) {
  const status = statusFromMastery(value);
  const cls =
    status === "green"
      ? "bg-emerald-400"
      : status === "amber"
      ? "bg-amber-400"
      : "bg-rose-400";
  return (
    <div className="rounded-2xl border border-slate-700/60 bg-slate-900/40 p-4">
      <div className="flex items-center justify-between">
        <div className="text-sm font-medium text-slate-100">{label}</div>
        <span className={`rounded-full border px-2 py-1 text-xs ${statusPill(status)}`}>{fmtPct(value)}</span>
      </div>
      <div className="mt-3 h-2 w-full rounded-full bg-slate-800">
        <div className={`h-2 rounded-full ${cls}`} style={{ width: `${clampPct(value)}%` }} />
      </div>
    </div>
  );
}

function HeatRow({ name, mastery, probability }: { name: string; mastery: number; probability: number }) {
  const status = statusFromMastery(mastery);
  const probBand = probability >= 0.85 ? "High" : probability >= 0.7 ? "Med" : "Low";
  const risk = probability >= 0.85 && mastery < 70;
  return (
    <div className="grid grid-cols-12 items-center gap-2 rounded-xl border border-slate-700/50 bg-slate-950/40 p-3">
      <div className="col-span-6">
        <div className="text-sm font-medium text-slate-100">{name}</div>
        <div className="text-xs text-slate-400">Probability band: {probBand}</div>
      </div>
      <div className="col-span-4">
        <div className="h-2 w-full rounded-full bg-slate-800">
          <div
            className={
              status === "green"
                ? "h-2 rounded-full bg-emerald-400"
                : status === "amber"
                ? "h-2 rounded-full bg-amber-400"
                : "h-2 rounded-full bg-rose-400"
            }
            style={{ width: `${clampPct(mastery)}%` }}
          />
        </div>
      </div>
      <div className="col-span-2 flex justify-end">
        <span className={`rounded-full border px-2 py-1 text-xs ${statusPill(status)}`}>{fmtPct(mastery)}</span>
        {risk ? <span className="ml-2 rounded-full bg-rose-500/20 px-2 py-1 text-xs text-rose-200">High Risk</span> : null}
      </div>
    </div>
  );
}

function RecommendationPanel({ snapshot }: { snapshot: SubjectSnapshot }) {
  const highProb = snapshot.topics
    .filter((t) => t.probability >= 0.85)
    .sort((a, b) => b.probability - a.probability);

  const weakestHighProb = highProb
    .map((t) => ({ ...t }))
    .sort((a, b) => a.mastery - b.mastery)[0];

  const weakestOverall = snapshot.topics
    .map((t) => ({ ...t }))
    .sort((a, b) => a.mastery - b.mastery)[0];

  const essayLow = snapshot.avgEssays < 65;
  const casesLow = snapshot.avgCases < 65;

  const recs: { title: string; desc: string; cta: string }[] = [];

  if (weakestHighProb && weakestHighProb.mastery < 70) {
    recs.push({
      title: `High Priority Fix: ${weakestHighProb.name}`,
      desc: `This topic is high-probability across DBE patterns. Your mastery is ${fmtPct(weakestHighProb.mastery)}.`,
      cta: "Run 25-mark Case Burst",
    });
  }
  if (essayLow) {
    recs.push({
      title: "Essay Upgrade",
      desc: `Your essay average is ${fmtPct(snapshot.avgEssays)}. Improve structure and insight scoring.`,
      cta: "Start Essay Planner",
    });
  }
  if (casesLow) {
    recs.push({
      title: "Application Boost",
      desc: `Case study average is ${fmtPct(snapshot.avgCases)}. Shift into timed application drills.`,
      cta: "Start Timed Case Drill",
    });
  }
  if (!recs.length && weakestOverall) {
    recs.push({
      title: `Next Best Move: ${weakestOverall.name}`,
      desc: `Your lowest topic mastery is ${fmtPct(weakestOverall.mastery)}. Fixing it lifts your overall readiness fast.`,
      cta: "Start Micro Drill",
    });
  }

  return (
    <div className="rounded-2xl border border-slate-700/60 bg-slate-900/40 p-4">
      <div className="flex items-center justify-between">
        <div className="text-sm font-semibold text-slate-100">Recommended Next Actions</div>
        <span className="text-xs text-slate-400">Adaptive engine</span>
      </div>
      <div className="mt-3 grid gap-3">
        {recs.map((r) => (
          <div key={r.title} className="rounded-xl border border-slate-700/60 bg-slate-950/40 p-3">
            <div className="text-sm font-semibold text-slate-100">{r.title}</div>
            <div className="mt-1 text-xs text-slate-400">{r.desc}</div>
            <div className="mt-2">
              <button className="rounded-xl bg-indigo-600 px-3 py-2 text-xs font-semibold text-white hover:bg-indigo-500">
                {r.cta}
              </button>
            </div>
          </div>
        ))}
      </div>
    </div>
  );
}
